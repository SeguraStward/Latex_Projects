\begin{frame}{Desafíos para la Adopción}
\framesubtitle{Barreras técnicas y de confianza}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{alertblock}{\faExclamationTriangle\; Sesgo Algorítmico}
Datos no diversos pueden perpetuar \textbf{desigualdades} en el diagnóstico.
\end{alertblock}
\vspace{3pt}
\begin{block}{\faServer\; Escalabilidad}
Alto costo de GPUs, escasez de datos anotados e infraestructuras hospitalarias \textbf{obsoletas}.
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{\faLock\; Caja Negra}
Sin explicabilidad no hay \textbf{confianza clínica} en el modelo.
\end{alertblock}
\vspace{3pt}
\begin{block}{\faSync\; Deriva de la IA}
El rendimiento se \textbf{degrada} con el tiempo al cambiar equipos o protocolos.
\end{block}
\end{column}
\end{columns}

\end{frame}
